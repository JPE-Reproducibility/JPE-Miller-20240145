\begin{frame}[label=movingAverage]{Difference-In-Differences}
	\begin{itemize}
		\item<1-> \textbf{Estimate single difference-in-differences specification:}
		\begin{align*}
		\textrm{Excess Returns}_{c,t} = 	& \alpha_{c}+\alpha_{t} +\beta\mathbbm{1}_{c,t}\{\textrm{Post}\times\textrm{Catholic}\times\textrm{Autocracy}\}  +\mathbf{\omega}^\prime \textrm{Controls}_{c,t}+\epsilon_{c,t} .
		\end{align*}
		\item<2-> \textbf{Two symmetric sample windows:} One from 1946--1976; one from 1939--1983. Exclude treatment years from 1959--1963.
		\vspace{.25cm}		
		\item<2-> \textbf{Two control groups:} all countries and autocracies only. \hyperlink{table6<1>}{\beamerskipbutton{Balance tests}}
		\vspace{.25cm}		
		\item<3-> \textbf{Controls:} political, event, and macro controls.
		\vspace{.25cm}		
		\item<4-> \textbf{Risk-adjustment:} excess returns adjusted for global and continental risk.
		
	\end{itemize}
\end{frame}
