\subsection{\scshape Sources}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\begin{frame}{Data}
	\begin{itemize}
		\item<1-> \textbf{Equity data:} come from Global Financial Data, Jorda-Schularick-Taylor Macrohistory database, IBES Global, and Factset.
		\begin{itemize}
			\vspace{.10cm}
			\item<1->[--] Unbalanced panel of 90 countries that spans more than 200 years.
			\vspace{.10cm}
			\item<1->[--] Average of 65 years per country for dividend yields data.
		\end{itemize}
		\vspace{.5cm}	
		\item<2-> \textbf{Episodes of Regime Transformation (ERT) and Lindberg et al. (2018).}
		\vspace{.10cm}
		\begin{itemize}
		\item<2->[--] \tabOnecolOneND\ episodes last an average of \demEpisodeLength\ years for episodes for which I have asset prices.
		\vspace{.10cm}
		\item<2->[--] Other measures of democratization produce similar results.
		\end{itemize}
		\vspace{.5cm}	
		\item<3-> \textbf{ERT provides start and end dates:} key for examining asset prices.
		\vspace{.10cm}
		\begin{itemize}
			\item<3->[--] Only data that provides coverage of entire democratization episode.
		\vspace{.10cm}
			\item<3->[--] Also provides success or failure of democratization.
		\end{itemize}
	\end{itemize}

\end{frame}
